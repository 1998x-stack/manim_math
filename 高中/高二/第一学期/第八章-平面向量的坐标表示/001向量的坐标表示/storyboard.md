### 向量绘制函数
```python
def create_vector(start, end, color, label=""):
    """创建向量箭头"""
    vector = Arrow(
        start=axes.c2p(*start),
        end=axes.c2p(*end),
        color=color,
        buff=0,
        stroke_width=4,
        tip_length=0.2
    )
    return vector
```

### 分量投影
```python
def create_projection(point, axis='x'):
    """创建投影虚线"""
    if axis == 'x':
        # 竖直虚线到 x 轴
        projection = DashedLine(
            axes.c2p(point[0], point[1]),
            axes.c2p(point[0], 0),
            dash_length=0.1,
            color=COLOR_AUXILIARY
        )
    else:
        # 水平虚线到 y 轴
        projection = DashedLine(
            axes.c2p(point[0], point[1]),
            axes.c2p(0, point[1]),
            dash_length=0.1,
            color=COLOR_AUXILIARY
        )
    return projection
```
