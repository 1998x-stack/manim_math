### 通用动画模式
```python
def show_symbol(self, number, name, shape, example, explanation):
    # 标题
    title = Text(f"{number} {name}", ...).move_to(UP*6)
    self.play(Write(title), run_time=0.5)
    
    # 符号图形
    self.play(Create(shape), run_time=0.6)
    
    # 示例文字
    self.play(FadeIn(example), run_time=0.4)
    
    # 说明
    explain = Text(explanation, ...).move_to(DOWN*3)
    self.play(FadeIn(explain), run_time=0.5)
    
    # 理解停顿
    self.wait(0.8)
    
    # 清理
    self.play(FadeOut(VGroup(title, shape, example, explain)), run_time=0.4)
```
